while true
do
    clear

    echo "==========================="
    echo "        MENU GRUPOS"
    echo "==========================="
    echo "1) Ingresar Grupo"
    echo "2) Dar de baja Grupo"
    echo "3) Listar Grupos"
    echo "4) Consultar Grupo"
    echo "5) Ver alumnos de un Grupo"
    echo "6) Volver al menu principal"
    echo "==========================="
    echo "Ingresar opcion:"
    read op

    case $op in
    1)
        echo "Ingresar nombre del grupo:"
        read grupo

        echo "Ingresar orientacion:"
        read or

        echo "Ingresar salon:"
        read sal

        echo "Ingresar nombre de adscripto:"
        read ad

        echo "Ingresar año del grupo:"
        read ano

        # Verifica si ya existe
        existe=$(grep "^$grupo," Grupos.dat)

        if [ -n "$existe" ]
        then
            echo "Ese grupo ya existe."
        else
            echo "$grupo,$or,$sal,$ad,$ano" >> Grupos.dat
            echo "Grupo ingresado correctamente."
        fi

        echo "Presione Enter Para Volver Al Menu De Grupos..."
        read skip
        sh Grupos.sh
        ;;
    2)
        echo "Ingresar nombre del grupo a eliminar:"
        read grupo

        # sigue verificando si existe, aca si no existe
        existe=$(grep "^$grupo," Grupos.dat)

        if [ -z "$existe" ]
        then
            echo "El grupo no existe."
        else
            # Verificar si tiene alumnos
            cantidad=$(grep ",$grupo$" Alumnos.dat | wc -l)

            if [ $cantidad -gt 0 ]
            then
                echo "No se puede eliminar el grupo."
                echo "El grupo tiene $cantidad alumno(s) asignado(s)."
            else
                grep -v "^$grupo," Grupos.dat > Grupos.tmp
                mv Grupos.tmp Grupos.dat
                echo "Grupo eliminado correctamente."
            fi
        fi

        echo "Presione Enter Para Volver Al Menu De Grupos..."
        read skip
        sh Grupos.sh
        ;;
    3)
        echo "=============================="
        echo "      LISTADO DE GRUPOS"
        echo "=============================="

        if [ ! -s Grupos.dat ]
        then
            echo "No hay grupos registrados."
        else
            while IFS=',' read grupo or sal ad ano
            do
                cantidad=$(grep ",$grupo$" Alumnos.dat | wc -l)

                echo
                echo "Grupo: $grupo"
                echo "Orientacion: $or"
                echo "Salon: $sal"
                echo "Adscripto: $ad"
                echo "Año: $ano"
                echo "Alumnos: $cantidad / 15"
                echo "-------------------------------"
            done < Grupos.dat
        fi

        echo "Presione Enter Para Volver Al Menu De Grupos..."
        read skip
        sh Grupos.sh
        ;;
    4)
        echo "Ingresar nombre del grupo:"
        read grupo

        datos=$(grep "^$grupo," Grupos.dat)

        if [ -z "$datos" ]
        then
            echo "El grupo no existe."
        else
            IFS=',' read nombre or sal ad ano <<< "$datos"

            cantidad=$(grep ",$grupo$" Alumnos.dat | wc -l)

            echo
            echo "=============================="
            echo "        DATOS DEL GRUPO"
            echo "=============================="
            echo "Grupo: $nombre"
            echo "Orientacion: $or"
            echo "Salon: $sal"
            echo "Adscripto: $ad"
            echo "Año: $ano"
            echo "Cantidad de alumnos: $cantidad / 15"
        fi

        echo "Presione Enter Para Volver Al Menu De Grupos..."
        read skip
        sh Grupos.sh
        ;;
    5)
        echo "Ingresar nombre del grupo:"
        read grupo

        existe=$(grep "^$grupo," Grupos.dat)

        if [ -z "$existe" ]
        then
            echo "El grupo no existe."
        else
            echo
            echo "================================"
            echo " ALUMNOS DEL GRUPO: $grupo"
            echo "================================"

            alumnos=$(grep ",$grupo$" Alumnos.dat)

            if [ -z "$alumnos" ]
            then
                echo "El grupo no tiene alumnos."
            else
                while IFS=',' read ci ape nom nac tel ano gr
                do
                    echo "CI: $ci - $ape $nom - Año: $ano"
                done <<< "$alumnos"
            fi
        fi

        echo "Presione Enter Para Volver Al Menu De Grupos..."
        read skip
        sh Grupos.sh
        ;;
    6)
        clear
        exit
        sh ProyectoFinal.sh
        ;;


    *)
        echo "Opcion incorrecta."
        echo "Presione Enter Para Volver Al Menu De Grupos..."
        read skip
        sh Grupos.sh
        ;;

    esac
done

while true
do
    clear

    echo "========================="
    echo "       MENU ALUMNOS"
    echo "========================="
    echo "1) Ingresar Alumno"
    echo "2) Dar de baja Alumno"
    echo "3) Modificar telefono"
    echo "4) Consultar Alumno"
    echo "5) Listar Alumnos"
    echo "6) Listar alumnos de un año"
    echo "7) Volver al menu principal"
    echo "========================="
    echo "Ingresar opcion:"
    read op

    case $op in
    1)
        echo "Ingresar CI:"
        read ci

        # Comprobar CI repetida
        existe=$(grep "^$ci," Alumnos.dat)

        if [ -n "$existe" ]
        then
            echo "Ya existe un alumno con esa CI."
        else

            echo "Ingresar apellido:"
            read ape

            echo "Ingresar nombre:"
            read nom

            echo "Fecha de nacimiento:"
            read nac

            echo "Ingresar telefono:"
            read tel

            echo "Ingresar año a cursar:"
            read ano

            
            # materias pendientes del alumno, analiza y ve si se puede inscribir
            
            echo
            echo "¿El alumno tiene asignaturas pendientes?"
            echo "1) Si"
            echo "2) No"
            read pendiente

            if [ "$pendiente" = "1" ]
            then
                echo "No se puede inscribir al alumno."
            else
                echo
                echo "Buscando grupos disponibles..."

                encontrado=0

                while IFS=',' read grupo or sal ad anogrupo
                do
                    if [ "$anogrupo" = "$ano" ]
                    then
                        cantidad=$(grep ",$grupo$" Alumnos.dat | wc -l)

                        if [ $cantidad -lt 15 ]
                        then
                            encontrado=1
                            grupoelegido=$grupo
                            break
                        fi
                    fi
                done < Grupos.dat

                # por si no hay grupos creados
                if [ $encontrado -eq 0 ]
                then
                    echo
                    echo "No existe un grupo disponible para el año $ano."
                    echo "Todos los grupos pueden estar llenos."
                    echo "Debe crear un nuevo grupo."
                else

                    echo
                    echo "Grupo asignado: $grupoelegido"

                    echo "$ci,$ape,$nom,$nac,$tel,$ano,$grupoelegido" >> Alumnos.dat

                    echo
                    echo "Alumno ingresado correctamente."
                    echo "Grupo asignado: $grupoelegido"
                fi
            fi
        fi

        read -p "Presione ENTER para continuar..."
        read skip
        sh Alumnos.sh
        ;;
    2)
        echo "Ingresar CI del alumno:"
        read ci

        existe=$(grep "^$ci," Alumnos.dat)

        if [ -z "$existe" ]
        then
            echo "No existe un alumno con esa CI."
        else
            grep -v "^$ci," Alumnos.dat > Alumnos.tmp
            mv Alumnos.tmp Alumnos.dat

            echo "Alumno eliminado correctamente."
        fi

        read -p "Presione ENTER para continuar..."
         read skip
         sh Alumnos.sh
        ;;
    3)
        echo "Ingresar CI del alumno:"
        read ci

        datos=$(grep "^$ci," Alumnos.dat)

        if [ -z "$datos" ]
        then
            echo "No existe un alumno con esa CI."
        else

            echo "Ingresar nuevo telefono:"
            read nuevo

            IFS=',' read ci2 ape nom nac tel ano grupo <<< "$datos"

            grep -v "^$ci," Alumnos.dat > Alumnos.tmp

            echo "$ci2,$ape,$nom,$nac,$nuevo,$ano,$grupo" >> Alumnos.tmp

            mv Alumnos.tmp Alumnos.dat

            echo "Telefono modificado correctamente."
        fi

        echo "Presione Enter para Volver al Menu de Alumnos..."
        read skip
        sh Alumnos.sh
        ;;
    4)
        echo "Ingresar CI:"
        read ci

        datos=$(grep "^$ci," Alumnos.dat)

        if [ -z "$datos" ]
        then
            echo "No existe un alumno con esa CI."
        else

            IFS=',' read ci2 ape nom nac tel ano grupo <<< "$datos"

            echo
            echo "-----------------------------"
            echo "       DATOS DEL ALUMNO"
            echo "-----------------------------"
            echo "CI: $ci2"
            echo "Apellido: $ape"
            echo "Nombre: $nom"
            echo "Fecha de nacimiento: $nac"
            echo "Telefono: $tel"
            echo "Año: $ano"
            echo "Grupo: $grupo"
        fi

        read -p "Presione ENTER para continuar..."
        read skip
        sh Alumnos.sh
        ;;
    5)
        echo "-----------------------------"
        echo "       LISTADO ALUMNOS"
        echo "-----------------------------"

        if [ ! -s Alumnos.dat ]
        then
            echo "No hay alumnos registrados."
        else

            while IFS=',' read ci ape nom nac tel ano grupo
            do
                echo "CI: $ci | $ape $nom | Año: $ano | Grupo: $grupo"
            done < Alumnos.dat

        fi

        read -p "Presione ENTER para continuar..."
        read skip
        sh Alumnos.sh
        ;;
    6)
        echo "Ingresar año:"
        read ano

        echo
        echo "--------------------------------"
        echo " ALUMNOS DEL AÑO $ano"
        echo "--------------------------------"

        encontrados=0

        while IFS=',' read ci ape nom nac tel anoalumno grupo
        do
            if [ "$anoalumno" = "$ano" ]
            then
                echo "CI: $ci | $ape $nom | Grupo: $grupo"
                encontrados=1
            fi
        done < Alumnos.dat

        if [ $encontrados -eq 0 ]
        then
            echo "No hay alumnos registrados en ese año."
        fi

        echo "Presione Enter Para Volver Al Menu de Alumnos..."
        read skip
        sh Alumnos.sh
        ;;

    7)
        clear

        sh ProyectoFinal.sh
        ;;


    *)
